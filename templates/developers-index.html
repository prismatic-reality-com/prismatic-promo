From Zero to Intelligence in 5 Minutes

Get started with your first intelligence query using our TypeScript SDK 

import { PrismaticSDK } from '@prismatic-platform/sdk';

const sdk = new PrismaticSDK({
  apiKey: process.env.PRISMATIC_API_KEY,
  endpoint: '//api.prismatic-platform.org'
});

// OSINT domain analysis
const analysis = await sdk.osint.analyzeDomain('example.com');
console.log('Security Rating:', analysis.security_rating);
console.log('Risk Factors:', analysis.risk_factors);

// Real-time threat intelligence
const threats = await sdk.threat.analyzeIndicator('198.51.100.1', 'ip');
console.log('Threat Score:', threats.threat_score);
